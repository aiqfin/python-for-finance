\section{Series：带标签的一列数据}
%==============================================================

\begin{frame}[fragile]{创建Series}
	\begin{lstlisting}
s = pd.Series([1500.35, 128.60, 55.20],
              index=["600519", "000858", "601318"])
s
\end{lstlisting}
	\begin{lstlisting}
600519    1500.35
000858     128.60
601318      55.20
dtype: float64
\end{lstlisting}
	\begin{itemize}
		\item 左边是\textbf{索引}（index，相当于标签），右边是值。
		\item 不指定索引时，默认为 0, 1, 2, \ldots（像列表）。
		\item 索引可以是股票代码、日期、任何你需要的标签。
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{按标签取值与向量化}
	\begin{lstlisting}
s["600519"]            # 1500.35  按标签取值
s[["600519", "601318"]]  # 取多个

s * 1.05               # 所有值涨5%（向量化，继承自NumPy）
s[s > 100]             # 筛选出大于100的
\end{lstlisting}
	\begin{itemize}
		\item Series 支持 NumPy 式向量化运算与布尔筛选——旧知识直接复用。
		\item 取值时用的是\alert{标签}而不是位置：\texttt{s["600519"]}。
	\end{itemize}
	\begin{alertblock}{关键概念：对齐}
		两个 Series 运算时按\textbf{索引对齐}，而不是按位置：\\
		"茅台的价格"只和"茅台的权重"相乘，顺序打乱也不会错。
	\end{alertblock}
\end{frame}


%==============================================================
\begin{frame}[fragile]{关键概念：对齐——索引顺序不同怎么办？}
\small
两位同学的基础分与加分分别记录如下，每个标签代表一位同学。\\
\textbf{问题：}加分表的顺序反了，怎样把加分加给对应的同学？
\begin{lstlisting}
base = pd.Series([10, 20], index=["A", "B"])
bonus = pd.Series([1, 2], index=["B", "A"])
\end{lstlisting}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{pandas：按索引标签相加}
\begin{lstlisting}
print(base + bonus)
# A    12
# B    21
# dtype: int64
\end{lstlisting}
A 找到 A：$10+2=12$；\\
B 找到 B：$20+1=21$。
\end{column}
\begin{column}{0.48\textwidth}
\textbf{NumPy：按数组位置相加}
\begin{lstlisting}
base_np = np.array([10, 20])
bonus_np = np.array([1, 2])
print(base_np + bonus_np)
# [11 22]
\end{lstlisting}
第一个加第一个：$10+1=11$；\\
第二个加第二个：$20+2=22$。
\end{column}
\end{columns}
\begin{alertblock}{顺序与标签}
NumPy 按位置运算，需要事先统一数据顺序；pandas \textbf{自动按相同的索引标签匹配}，即使排列顺序不同，也能正确相加。
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Series 对齐：缺少一个索引标签会怎样？}
\small
沿用同学分数的例子：基础分 Series 有 A、B、C，\\
加分 Series 只有 B、A。\textbf{相加时，找不到对应加分的 C 会得到什么？}
\begin{lstlisting}
base = pd.Series([10, 20, 30],
                 index=["A", "B", "C"])
bonus = pd.Series([1, 2],
                  index=["B", "A"])
\end{lstlisting}
\begin{columns}[T]
\begin{column}{0.42\textwidth}
\begin{lstlisting}
print(base + bonus)
# A   12.0
# B   21.0
# C    NaN
# dtype: float64
\end{lstlisting}
\end{column}
\begin{column}{0.54\textwidth}
\textbf{自动匹配后的计算}
\begin{itemize}
\item A：$10+2=12$。
\item B：$20+1=21$。
\item C：基础分为 30，但没有对应加分，结果为 \texttt{NaN}（缺失值）。
\end{itemize}
\end{column}
\end{columns}
\begin{alertblock}{自动对齐不等于自动补零}
两个 Series 相加时\textbf{自动按索引标签匹配}，保留双方出现的标签；缺少对应值时，结果为 \texttt{NaN}，不会把缺失的加分自动当成 0。
\end{alertblock}
\end{frame}

\section{数据选取与筛选}
%==============================================================

\begin{frame}[fragile]{取列与取行}
	\begin{lstlisting}
df["茅台"]              # 取一列 -> Series
df[["茅台", "平安"]]    # 取多列 -> DataFrame
df["2026-07-01"]        # 报错！方括号默认是"按列"
\end{lstlisting}
	\begin{itemize}
		\item \texttt{df[...]} 里放列名：取\textbf{列}。
		\item 想按行取，用切片（按位置）或下一页的 \texttt{loc/iloc}。
	\end{itemize}
	\begin{lstlisting}
df[0:2]          # 前两行（位置切片，含头不含尾）
\end{lstlisting}
	\begin{alertblock}{最常见的困惑}
		为什么 \texttt{df["茅台"]} 可以，\texttt{df["2026-07-01"]} 却报错？\\
		因为 \texttt{[]} 默认针对\textbf{列}。规则记牢，少查一半报错。
	\end{alertblock}
\end{frame}

\begin{frame}[fragile]{loc：按标签选；iloc：按位置选}
	\begin{lstlisting}
df.loc["2026-07-02"]              # 该行全部列
df.loc["2026-07-02", "茅台"]      # 指定行、指定列 -> 1520.0
df.loc[:, "茅台"]                 # 所有行、茅台列

df.iloc[0]                        # 第0行（按位置）
df.iloc[0:2, 1]                   # 前2行、第1列
df.iloc[-1]                       # 最后一行
\end{lstlisting}
	\begin{itemize}
		\item \texttt{loc}：用\textbf{标签}（日期、列名）；写法 \texttt{[行, 列]}。
		\item \texttt{iloc}：用\textbf{位置}（0, 1, 2\ldots）；与列表索引规则相同。
		\item \alert{先想清楚"我要按名字找还是按位置找"，再选工具}。
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{条件筛选：选出满足条件的行}
	\begin{lstlisting}
df[df["茅台"] > 1510]       # 茅台价格高于1510的日子
df[(df["茅台"] > 1500) & (df["平安"] > 55)]
\end{lstlisting}
	\begin{itemize}
		\item 与 NumPy 布尔索引完全一致：先算出布尔列，再用来筛行。
		\item 多条件组合用 \texttt{\&}、\texttt{|}，每个条件加括号。
		\item 金融例子：涨幅超过5\%的日子、PE低于10且股息率高于4\%的股票。
	\end{itemize}
	\begin{exampleblock}{课堂练习}
		用一行代码选出"五粮液价格低于128"的所有交易日。
	\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{日期切片：先建立 DatetimeIndex}
\small 沿用三只股票的价格表。\textbf{问题：}怎样让 pandas 把索引理解为日期，并支持按月选取？
\begin{lstlisting}
df.index = pd.to_datetime(df.index)
df = df.sort_index()
print(type(df.index))
# <class 'pandas.core.indexes.datetimes.DatetimeIndex'>
print(df.index.strftime("%Y-%m-%d").tolist())
# ['2026-07-01', '2026-07-02', '2026-07-03']
\end{lstlisting}
\begin{itemize}
\item \texttt{pd.to\_datetime} 将日期字符串解析成日期时间，索引成为 \texttt{DatetimeIndex}。
\item \texttt{sort\_index()} 按日期升序排列；后面都在这个唯一、有序的日期索引上操作。
\item \texttt{strftime} 仅用于格式化显示，\texttt{df.index} 仍是日期时间索引。
\end{itemize}

\begin{frame}[fragile]{时间标签切片与位置切片}
\small \textbf{问题：}取 7 月 1 日到 3 日的茅台价格，与取第 0、1 个位置，有什么不同？
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{loc：按日期范围}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
print(df.loc["2026-07-01":"2026-07-03",
             "茅台"])
# 2026-07-01    1500.0
# 2026-07-02    1520.0
# 2026-07-03    1515.0
# Name: 茅台, dtype: float64
\end{lstlisting}
包括起止日期，得到 3 行。
\end{column}
\begin{column}{0.48\textwidth}
\textbf{iloc：按当前位置}
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
print(df.iloc[0:2, 0])
# 2026-07-01    1500.0
# 2026-07-02    1520.0
# Name: 茅台, dtype: float64
\end{lstlisting}
包括位置 0，不包括位置 2，得到 2 行。
\end{column}
\end{columns}
\begin{block}{日期范围不是行数}
\texttt{loc} 根据时间标签确定范围；\texttt{iloc} 根据当前行的位置取值。日期切片只返回已有记录，不会补出周末、节假日或缺失日期。
\end{block}
\end{frame}

\end{itemize}

\section{排序与分组}
\begin{frame}[fragile]{排序：先按股票，再按日期}
\small \textbf{问题：}怎样把同一只股票的记录放在一起，并按时间先后排列？
\begin{lstlisting}
market = market.sort_values(["股票", "日期"])
print(market.head(3))
#     股票         日期    收盘价
# 6  五粮液 2026-07-01  128.0
# 7  五粮液 2026-07-02  129.5
# 8  五粮液 2026-07-03  127.0
\end{lstlisting}
\begin{itemize}
\item 排序键按先后生效：先按股票名排序，同一股票内再按日期升序。
\item 日期在这里是\textbf{普通列}，所以用 \texttt{sort\_values}；\texttt{sort\_index} 排的是左侧行索引。
\item 输出左侧的 6、7、8 是保留的原行索引，不是重新编号。
\item 排序返回新表，通过赋值更新 \texttt{market}。后面的首末价格计算都使用这一日期顺序。
\end{itemize}

\begin{frame}[fragile]{groupby：分组、选列、聚合}
\small \textbf{问题：}每只股票在已有记录中的平均收盘价是多少？
\begin{lstlisting}
print(market.groupby("股票")["收盘价"].mean().round(2))

# 五粮液     129.12
# 平安        55.17
# 茅台      1522.50
# Name: 收盘价, dtype: float64
\end{lstlisting}
\begin{enumerate}
\item \texttt{groupby("股票")}：同一股票在不同日期的记录归入一组。
\item \texttt{["收盘价"]}：指定要计算的数值列。
\item \texttt{mean()}：各组分别求均值；\texttt{round(2)} 将结果保留两位小数。
\end{enumerate}
\begin{block}{从每日记录到股票汇总}
输入 15 行行情，输出 3 个股票均值。茅台、五粮液各用 6 天，平安仅用 3 天；这里是各自已有记录的价格均值。
\end{block}
\end{frame}

\begin{frame}[fragile]{分组循环：每次拿到一只股票的小表}
\begin{lstlisting}
for name, group in market.groupby("股票", sort=False):
    print(name, len(group),
          group["收盘价"].iloc[0],
          group["收盘价"].iloc[-1])
# 五粮液 6 128.0 131.2
# 平安 3 55.0 56.0
# 茅台 6 1500.0 1545.0
\end{lstlisting}
\begin{itemize}
\item \texttt{name} 是组名，例如“茅台”；\texttt{group} 是该股票全部记录组成的 DataFrame，保留股票、日期、收盘价三列。
\item 每行输出依次为：股票名、记录数、首日价、末日价。
\item \texttt{sort=False} 按组名首次出现的顺序遍历；组内行顺序沿用输入，本节开头已按股票和日期排序。
\end{itemize}
\begin{block}{什么时候用循环？}
适合逐组查看数据、画图或保存结果。只是计算均值、最大值等常规统计时，直接使用聚合方法更简洁。
\end{block}
\end{frame}

\begin{lstlisting}
kept = market.groupby("股票", sort=False).filter(
    lambda group: len(group) >= 5)
print(kept.groupby("股票", sort=False).size())
# 股票
# 五粮液    6
# 茅台      6
# dtype: int64
print(kept.shape)
# (12, 3)
\end{lstlisting}
\begin{itemize}
\item \texttt{lambda group: ...} 接收一组数据，返回一个布尔值：这一组是否满足条件。
\item 茅台、五粮液各有 6 条，保留全部记录；平安仅 3 条，整组删除。
\item 输出仍是原来的三列、共 12 行行情。上方的 \texttt{size()} 只是用来检查筛选结果。
\end{itemize}
\textbf{区别：}布尔索引按每行的条件筛选；\texttt{GroupBy.filter} 按整组的条件决定去留。本例数的是记录行数，不是验证所有应有交易日都已齐全。
\end{frame}

\begin{frame}[fragile]{apply：每组执行自定义函数}
\small \textbf{问题：}把“取首末价格并计算涨幅”写成函数，如何分别应用于保留下来的股票？
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
def price_summary(group):
    prices = group["收盘价"]
    return pd.Series({
        "首日价": prices.iloc[0],
        "末日价": prices.iloc[-1],
        "区间涨幅(%)": (prices.iloc[-1] / prices.iloc[0] - 1) * 100})

result = kept.groupby("股票", sort=False)[["收盘价"]].apply(
    price_summary)
print(result.round(2))
\end{lstlisting}
\begin{center}
\small
\begin{tabular}{lrrr}
\toprule
股票 & 首日价 & 末日价 & 区间涨幅(\%) \\
\midrule
五粮液 & 128.0 & 131.2 & 2.5 \\
茅台 & 1500.0 & 1545.0 & 3.0 \\
\bottomrule
\end{tabular}
\end{center}
\small 每组传入一列价格组成的 DataFrame；函数返回带指标名称的 Series，\texttt{apply} 将它们合成汇总表。本例两组均覆盖 7 月 1--8 日；涨幅是首末价格变化，未计分红与交易成本。
\end{frame}

\begin{frame}[fragile]{同一项计算：优先考虑内置聚合}
\small 自定义 \texttt{apply} 很灵活；如果计算能拆成现成的聚合方法，也可以直接写：
\begin{lstlisting}
result = kept.groupby("股票", sort=False).agg(
    首日价=("收盘价", "first"),
    末日价=("收盘价", "last"))
result["区间涨幅(%)"] = (
    result["末日价"] / result["首日价"] - 1) * 100
print(result.round(2))
\end{lstlisting}
\begin{center}
\small
\begin{tabular}{lrrr}
\toprule
股票 & 首日价 & 末日价 & 区间涨幅(\%) \\
\midrule
五粮液 & 128.0 & 131.2 & 2.5 \\
茅台 & 1500.0 & 1545.0 & 3.0 \\
\bottomrule
\end{tabular}
\end{center}
\small 本例价格无缺失，两种写法结果相同。\texttt{first/last} 默认取组内首末非缺失值，而 \texttt{iloc[0]/iloc[-1]} 取实际首末位置；有缺失时需明确口径。两种方法都依赖事先排好的日期顺序。

\begin{frame}[fragile]{运算时补零：先确认缺失的含义}
\small \textbf{问题：}合并两个账户的持股数量。已确认清单完整，未列出的股票表示该账户持有 0 股。
\begin{lstlisting}
account_a = pd.Series({"茅台": 100, "平安": 200})
account_b = pd.Series({"平安": 300, "招行": 400})
total = account_a.add(account_b, fill_value=0)
print(total.reindex(["茅台", "平安", "招行"]))
# 茅台    100.0
# 平安    500.0
# 招行    400.0
# dtype: float64
\end{lstlisting}
\begin{itemize}
\item \texttt{fill\_value=0} 在相加前，将一方缺少的对应值按 0 处理。
\item 茅台：$100+0=100$ 股；平安：$200+300=500$ 股；招行：$0+400=400$ 股。
\item \texttt{reindex} 仅指定结果展示的股票顺序。
\end{itemize}
\begin{alertblock}{缺失价格不能照此处理}
未取得报价表示价格未知，不表示价格为 0。只有业务含义支持时才补零；双方都缺失的位置仍为缺失值。
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{1. 两张表的粒度与键}
\small 合成订单表每行一笔订单，商品表每行一种商品。同一商品可以出现在多笔订单中，因此合并关系是多对一。

products = pd.DataFrame({
    "商品": ["T", "C", "J"],
    "名称": ["茶", "咖啡", "果汁"], "单价": [10, 20, 15]})
print(orders)
print(products)

\begin{frame}[fragile]{6. concat 的另一方向：横向按索引对齐}
\small axis=1 横向拼接时按索引标签对齐，不会因为行数相同就按位置贴在一起。先检查索引含义。
\medskip
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
left = pd.Series([80, 90], index=["001", "002"], name="数学")

\begin{frame}{案例任务：从行情表到可比的收益率样本}
沿用茅台、五粮液、平安在 7 月 1--8 日的教学行情，不另建数据集。
\begin{itemize}
\item \texttt{prices\_wide}：6 个日期 $\times$ 3 只股票，保留缺失价格。
\item \texttt{long\_prices}：18 行“日期--股票”组合，包括平安后 3 天的缺失报价。
\item \texttt{market}：15 条有效报价，适合查看已知价格，但不保留完整的日期组合。
\end{itemize}
\begin{block}{要完成的分析}
检查每天的数据覆盖情况；在各只股票内部计算相邻观测日收益率；整理收益率矩阵，并确定三只股票可共同比较的日期。

\small 本案例使用 \texttt{long\_prices} 保留缺失位置。提前删掉缺失价格后再错位，可能把跨多日的价格变化误认为相邻观测日变化。

\begin{frame}{综合案例：整理成绩登记表}
\begin{itemize}
\item 输入：6 条原始登记，包含文本差异、重复、缺考、异常分数和待核日期。
\item 任务：应用字符串处理、类型转换、重复检查、异常标记及缺失处理，形成可解释的统计结果。
\item 输出：保留记录、有效成绩、问题标记与清洗汇总；最后检查导出再读回是否保留学号。
\end{itemize}
\begin{block}{练习方式}
先根据原始表判断问题，再逐步运行并核对结果。配套练习与折叠参考答案见 \texttt{04\_pandas} 第12节。
\end{block}
\end{frame}
\begin{frame}[fragile]{1. 原始登记表：保留文本才能看见问题}
\small 教学合成数据，每行一次成绩登记。故意保留首尾空格、重复记录、缺考和不合理成绩；学号按字符串读入以保留前导零。
\medskip
\begin{lstlisting}
from io import StringIO
import pandas as pd
raw_csv = """学号,姓名,班级,成绩,日期
001, 小林 ,A,82,2026-09-01
002,小周,a,95,2026-09-01
003,小陈,B,缺考,2026-09-02
004,小许,B,108,2026-09-02
001, 小林 ,A,82,2026-09-01
005,小吴, B ,76,日期待核"""
raw = pd.read_csv(StringIO(raw_csv), dtype="string")
raw
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{2. 先检查，再清洗}
\small info 看类型和非缺失数，duplicated 找整行重复。这里“缺考”是普通文本，尚不会被 isna 自动识别。
\medskip
\begin{lstlisting}
raw.info()
print(raw.isna().sum())
print("整行重复数：", raw.duplicated().sum())
raw.loc[raw.duplicated(keep=False)]
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{3. str 方法：把单个字符串操作扩展到整列}
\small copy 保留原始登记。统一班级格式后再分组，否则 A 和 a 会被算成两个班。
\medskip
\begin{lstlisting}
clean = raw.copy()
clean["姓名"] = clean["姓名"].str.strip()
clean["班级"] = clean["班级"].str.strip().str.upper()
print(clean["班级"].value_counts())
print(clean["姓名"].str.contains("小", na=False))
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{4. 类型转换：转换失败的行必须可追踪}
\small 先把明确的缺考编码改成缺失，再转数值。coerce 将无法解析的值变成缺失；保留原列才能区分缺考与格式错误。
\medskip
\begin{lstlisting}
score_text = clean["成绩"].replace("缺考", pd.NA)
clean["分数"] = pd.to_numeric(score_text, errors="coerce")
clean["登记日"] = pd.to_datetime(clean["日期"], errors="coerce")
parse_failed = score_text.notna() & clean["分数"].isna()
print(clean.loc[parse_failed | clean["登记日"].isna()])
print(clean.dtypes)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{5. 重复键不等于重复行}
\small 本例每名学生只有一次考试，学号应唯一。已确认整行相同的重复登记可以删除；若同一学号分数不同，应先核实，不能任意保留。
\medskip
\begin{lstlisting}
print(clean.loc[clean.duplicated("学号", keep=False)])
clean = clean.drop_duplicates().reset_index(drop=True)
print("清理后记录数：", len(clean))
print("学号唯一：", clean["学号"].is_unique)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{6. 异常值与缺失值分开记录}
\small 考试满分 100，因此 108 是待核异常，不能直接截成 100。缺考也不等于 0 分。设置标记后，在统计列中排除异常。
\medskip
\begin{lstlisting}
clean["成绩异常"] = (clean["分数"].notna() &
                       ~clean["分数"].between(0, 100))
clean["有效成绩"] = clean["分数"].mask(clean["成绩异常"])
print(clean[["学号", "成绩", "成绩异常", "有效成绩"]])
print("有效成绩均值：", clean["有效成绩"].mean())
\end{lstlisting}
\end{frame}

计算有效成绩数量、缺考数量、异常成绩数量。为什么不能把三者混成一类？
\medskip

\begin{frame}[fragile]{7. 删除与填充：输出改变了什么}
\small 比较同一列的有效值平均与填零平均。dropna 的 subset 明确本次分析依赖哪些列；不能因为日期缺失就删除不依赖日期的成绩。
\medskip
\begin{lstlisting}
print(clean["有效成绩"].mean())
print(clean["有效成绩"].fillna(0).mean())
usable_scores = clean.dropna(subset=["有效成绩"])
print(usable_scores[["学号", "有效成绩"]])
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{8. 分类与映射：明确区间边界}
\small cut 按固定阈值分段，right=False 表示左闭右开。100 包含在 [90,101) 内；没有有效成绩的行仍保留缺失等级。
\medskip
\begin{lstlisting}
clean["等级"] = pd.cut(clean["有效成绩"],
    bins=[0, 60, 80, 90, 101], right=False,
    labels=["待提高", "合格", "良好", "优秀"])
clean["班级名称"] = clean["班级"].map({"A": "一班", "B": "二班"})
clean[["学号", "有效成绩", "等级", "班级名称"]]
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{9. 导出与读回：CSV 不保存所有类型}
\small 先在内存中演示 CSV 往返。读回时显式指定学号类型；分类类型和日期类型需要重建。真实文件路径可替换 StringIO。
\medskip
\begin{lstlisting}
csv_text = clean.to_csv(index=False)
restored = pd.read_csv(StringIO(csv_text), dtype={"学号": "string"})
print(restored["学号"].tolist())
print(len(raw), len(clean), len(restored))
assert len(raw) == 6 and len(clean) == 5
assert restored["学号"].iloc[0] == "001"
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{10. 案例汇总：给出可解释的清洗记录}

\begin{frame}[fragile]{综合案例：三股风险分析（构造数据）}
	\begin{lstlisting}
rng = np.random.default_rng(42)
dates = pd.date_range("2026-01-05", periods=250, freq="B")

# 模拟三只股票的日对数收益率，再还原为价格
rets = pd.DataFrame({
    "茅台": 0.0004 + 0.015 * rng.standard_normal(250),
    "平安": 0.0002 + 0.020 * rng.standard_normal(250),
    "招行": 0.0003 + 0.018 * rng.standard_normal(250)},
    index=dates)
prices = 100 * np.exp(rets.cumsum())
prices.tail(3)          # 查看最后3天
\end{lstlisting}
	\begin{itemize}
		\item 用 DataFrame 字典一次性构造三列，索引是简化的工作日序列。
		\item \texttt{cumsum}、\texttt{exp} 对整张表向量化生效——NumPy 功力再现。
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{综合案例：收益率、风险与相关性}
	\begin{lstlisting}
rets = prices.pct_change(fill_method=None).dropna()   # 日收益率

rets.describe().round(4)              # 统计摘要
rets.mean() * 252                     # 年化平均收益
rets.std() * np.sqrt(252)             # 年化波动率
rets.corr().round(2)                  # 相关系数矩阵
\end{lstlisting}
	\begin{itemize}
		\item 一个方法作用于\textbf{所有列}：这就是 DataFrame 的威力。
		\item \textbf{相关系数矩阵}：衡量资产间联动性，组合分散化的依据。
		\item 从构造数据到风险报告，没有写一个循环。
	\end{itemize}

\begin{frame}{本章小结}
	\begin{itemize}
		\item \textbf{两种结构}：Series（一列带标签）、DataFrame（一张表）；\alert{行=日期，列=资产}。
		\item \textbf{选取}：\texttt{df[列名]} 取列；\texttt{loc} 按标签、\texttt{iloc} 按位置；布尔条件筛行。
		\item \textbf{排序分组}：\texttt{sort\_values/sort\_index}；\texttt{groupby} 三步走。
		\item \textbf{时间序列}：\texttt{to\_datetime}、\texttt{pct\_change}、\texttt{shift}、\texttt{resample}。
		\item \textbf{缺失值}：\texttt{isna/dropna/fillna/ffill}。
		\item \textbf{读写}：\texttt{read\_csv(index\_col=0, parse\_dates=True)}。
		\item \textbf{金融工具箱}：\texttt{describe()}、年化波动率 $\sigma\times\sqrt{252}$、相关矩阵 \texttt{corr()}。
	\end{itemize}
\end{frame}
