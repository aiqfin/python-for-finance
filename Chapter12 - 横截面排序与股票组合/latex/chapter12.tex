\section{组合权重与收益计算}
\begin{frame}{相同股票分组，两种初始权重}
\begin{columns}[T]
\begin{column}{0.46\textwidth}
\begin{block}{等权}
\[w_i=\frac{1}{N}\]
每只股票投入同样金额。\par
小股票在组合中也获得相同份额。
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{市值加权}
\[w_i=\frac{ME_{i,s}}{\sum_j ME_{j,s}}\]
按信号日市值分配金额。\par
组内较大股票获得更高权重。
\end{block}
\end{column}
\end{columns}
\vspace{0.4cm}
全体候选股票按相同方式构建\texttt{ALL}基准。\par
这是同池自建组合，不是官方市场指数。
\nbref{第4.1节}
\end{frame}

\begin{frame}{买入持有：权重会自然变化}
\centering
\begin{tabular}{lrrr}
\toprule & 建仓日 & 第1天 & 第2天\\\midrule
股票A价格 & 100 & 110 & 121\\
股票B价格 & 100 & 100 & 100\\
买入持有净值 & 1.0000 & 1.0500 & 1.1050\\
每日恢复等权净值 & 1.0000 & 1.0500 & 1.1025\\\bottomrule
\end{tabular}
\vspace{0.45cm}\par\raggedright
初始各投0.5。第1天后A的权重变为$0.55/1.05$，已经高于50\%。\par
每天对个股收益取简单平均，相当于每天恢复等权，改变了调仓规则。
\nbref{第4节；Notebook含可运行例子与预期终值核对}
\end{frame}

\begin{frame}{用可比价格计算整段持有价值}
\[V_{g,t}=\sum_i w_{i,g}\frac{P^*_{i,t}}{P^*_{i,e}}\]
\[r_{g,t}=\frac{V_{g,t}}{V_{g,t-1}}-1\]
\begin{itemize}
\item $P^*$：含红利可比价格；$e$：建仓日。
\item $w_{i,g}$：本期初始股票权重，各组权重合计为1。
\item 建仓日价值为1；之后的变化归新组合。
\item 月末更新持仓，下一期重复计算并串联日收益。
\end{itemize}
\nbref{第4.2节；无论等权还是市值加权，使用同一估值公式}
\end{frame}

\begin{frame}[fragile]{矩阵计算：日期乘股票，股票乘组合}
\begin{lstlisting}[basicstyle=\ttfamily\small]

\end{lstlisting}
\begin{center}
\begin{tabular}{lll}

\texttt{relative} & 日期 & 股票\\
\texttt{weights} & 股票 & 组合\\
\texttt{values} & 日期 & 组合\\\bottomrule
\end{tabular}
\end{center}
\nbref{第4.2节；按股票标签对齐，先理解维度再使用矩阵乘法}
\end{frame}

\begin{frame}{统一价格表，集中学习组合计算}
\begin{itemize}
\item 将含红利可比价格转成日期乘股票的矩阵，按市场交易日对齐。
\item 使用\texttt{ffill()}沿用历史最后有效价格，之后直接用于计算。
\item 信号日已有正的有效估值，保证建仓价格比值可以计算。
\item 各组全部资金投入股票，月内买入持有、月末重新分组。
\end{itemize}
\begin{block}{教学假设}
调仓日按当日估值完成全部买卖，不考虑停牌、涨跌停和成交限制。
\end{block}
\nbref{第2.1、4节；只用过去价格填充，不向前引用未来价格}
\end{frame}

\section{结果与敏感性}
\begin{frame}{从组合收益到风险收益指标}
\begin{align*}
\text{累计收益}&=\prod_{t=1}^{T}(1+r_t)-1,\\
\text{几何年化收益}&=\left[\prod_{t=1}^{T}(1+r_t)\right]^{252/T}-1,\\
\text{年化波动率}&=s(r_t)\sqrt{252},\\
\text{回撤}_t&=\frac{NAV_t}{\max_{0\leq u\leq t}NAV_u}-1.
\end{align*}
\begin{itemize}
\item 252用于近似年化，回撤的历史峰值包含初始净值1。
\item 没有无风险利率，本章不报告Sharpe比率或alpha。
\end{itemize}
\nbref{第6节；\ReturnDays{}个收益交易日}
\end{frame}

\section{选学：换手与成本}
\begin{frame}{选学：在组合结果之后计算成本}
\[\mathrm{Turnover}=\tfrac12\sum_i|w_i^{new}-w_i^{old}|\]
\[\mathrm{Cost}=0.001\sum_i|w_i^{new}-w_i^{old}|\]
\begin{itemize}
\item 上式用于已有持仓的月度调仓；旧权重取自然漂移后的期末权重。
\item 首次全额建仓的买入金额为1，换手单独记为1。
\item 每买入或卖出1元，假设支付10bp成本。
\item 在新一期首个收益日比例扣费，作为独立选学情景。
\end{itemize}
\nbref{第11节；主分析无需运行本节，扣费使用毛组合权重估算交易金额}
\end{frame}
